<script>
  const player = document.getElementById('player');
  const scoreEl = document.getElementById('score');
  const holdingEl = document.getElementById('holding');
  const timerEl = document.getElementById('timer');
  const container = document.getElementById('game-container');
  const gameOverEl = document.getElementById('game-over');
  const finalScoreEl = document.getElementById('final-score');

  // Posição e velocidade
  let playerPos = { x: 420, y: 250 };
  const speed = 9; // Velocidade levemente aumentada para compensar o tamanho maior
  const keys = {};

  let currentItem = null;
  let score = 0;
  let activeItems = [];
  let timeLeft = 60;
  let gameInterval = null;
  let timerInterval = null;
  let isGameOver = false;

  // Lista de resíduos
  const itemsData = [
    // Vidro (Verde)
    { type: 'vidro', img: 'vidro_1_.png', name: 'Garrafa de Vidro' },
    { type: 'vidro', img: 'vidro_3_.png', name: 'Frasco de Vidro' },
    { type: 'vidro', img: 'vidro_4_.png', name: 'Potes de Vidro' },
    // Papel (Azul)
    { type: 'papel', img: 'caixa_1_.png', name: 'Caixa de Papelão' },
    { type: 'papel', img: 'papelao_2_.png', name: 'Papelão' },
    // Orgânico (Marrom)
    { type: 'organico', img: 'abacate_.png', name: 'Abacate' },
    { type: 'organico', img: 'banana_.png', name: 'Banana' },
    { type: 'organico', img: 'cha_.png', name: 'Saco de Chá' },
    { type: 'organico', img: 'peixe_.png', name: 'Espinha de Peixe' },
    { type: 'organico', img: 'melanci_.png', name: 'Melancia' },
    { type: 'organico', img: 'casca_de_ovo_.png', name: 'Casca de Ovo' },
    { type: 'organico', img: 'resto_de_maça_.png', name: 'Maçã' }
  ];

  // Controles por teclado
  window.addEventListener('keydown', e => keys[e.key] = true);
  window.addEventListener('keyup', e => keys[e.key] = false);

  // Gerenciador do Tempo
  function startTimer() {
    timerInterval = setInterval(() => {
      timeLeft--;
      timerEl.innerText = timeLeft;

      if (timeLeft <= 0) {
        endGame();
      }
    }, 1000);
  }

  function endGame() {
    isGameOver = true;
    clearInterval(timerInterval);
    clearInterval(gameInterval);
    finalScoreEl.innerText = score;
    gameOverEl.style.display = 'flex';
  }

  function restartGame() {
    // Reset de variáveis
    score = 0;
    timeLeft = 60;
    isGameOver = false;
    currentItem = null;
    playerPos = { x: 420, y: 250 };

    scoreEl.innerText = score;
    timerEl.innerText = timeLeft;
    holdingEl.innerText = 'Nenhum';
    gameOverEl.style.display = 'none';

    // Remover itens da tela
    activeItems.forEach(item => item.element.remove());
    activeItems = [];

    // Reiniciar loops
    startTimer();
    gameInterval = setInterval(spawnItem, 2500);
    requestAnimationFrame(update);
  }

  // Aparecer itens
  function spawnItem() {
    if (isGameOver || activeItems.length >= 4) return;

    const randomData = itemsData[Math.floor(Math.random() * itemsData.length)];
    const itemEl = document.createElement('div');
    itemEl.classList.add('item');
    itemEl.style.backgroundImage = `url('${randomData.img}')`;
    
    const x = Math.floor(Math.random() * 840) + 40;
    const y = Math.floor(Math.random() * 260) + 100;
    
    itemEl.style.left = x + 'px';
    itemEl.style.top = y + 'px';

    container.appendChild(itemEl);

    activeItems.push({
      element: itemEl,
      type: randomData.type,
      name: randomData.name,
      x: x,
      y: y
    });
  }

  // Loop Principal
  function update() {
    if (isGameOver) return;

    // Movimentação ajustada para o personagem maior de 160px
    if (keys['ArrowUp'] || keys['w'] || keys['W']) {
      playerPos.y = Math.max(40, playerPos.y - speed);
      player.style.backgroundImage = "url('personagem_costas_.png')";
    }
    if (keys['ArrowDown'] || keys['s'] || keys['S']) {
      playerPos.y = Math.min(480, playerPos.y + speed);
      player.style.backgroundImage = "url('personagem_frente_.png')";
    }
    if (keys['ArrowLeft'] || keys['a'] || keys['A']) {
      playerPos.x = Math.max(-20, playerPos.x - speed);
      player.style.backgroundImage = "url('personagem_lado_esquerdo_.png')";
    }
    if (keys['ArrowRight'] || keys['d'] || keys['D']) {
      playerPos.x = Math.min(860, playerPos.x + speed);
      player.style.backgroundImage = "url('personagem_lado_direito_.png')";
    }

    player.style.left = playerPos.x + 'px';
    player.style.top = playerPos.y + 'px';

    // Centro do personagem para cálculo de colisão
    const playerCenterX = playerPos.x + 80;
    const playerCenterY = playerPos.y + 80;

    // Colisão / Coleta de itens
    if (!currentItem) {
      activeItems.forEach((item, index) => {
        const itemCenterX = item.x + 30;
        const itemCenterY = item.y + 30;

        if (Math.abs(playerCenterX - itemCenterX) < 70 && Math.abs(playerCenterY - itemCenterY) < 70) {
          currentItem = item;
          item.element.remove();
          activeItems.splice(index, 1);
          holdingEl.innerText = item.name;
        }
      });
    }

    // Colisão com as lixeiras na parte inferior
    if (currentItem && playerPos.y > 380) {
      // Lixeira Verde (Vidro)
      if (Math.abs(playerCenterX - 200) < 70) {
        checkDeposit('vidro');
      } 
      // Lixeira Azul (Papel)
      else if (Math.abs(playerCenterX - 500) < 70) {
        checkDeposit('papel');
      } 
      // Lixeira Marrom (Orgânico)
      else if (Math.abs(playerCenterX - 800) < 70) {
        checkDeposit('organico');
      }
    }

    requestAnimationFrame(update);
  }

  function checkDeposit(targetType) {
    if (currentItem.type === targetType) {
      score += 10;
    } else {
      score = Math.max(0, score - 5);
    }
    scoreEl.innerText = score;
    currentItem = null;
    holdingEl.innerText = 'Nenhum';
  }

  // Inicialização do Jogo
  startTimer();
  gameInterval = setInterval(spawnItem, 2500);
  update();
</script>
